import yfinance as yf import pandas as pd from datetime import datetime import firebase_admin from firebase_admin import credentials, firestore # Firebase Initialize if not firebase_admin._apps: cred = credentials.Certificate("firebase_credentials.json") firebase_admin.initialize_app(cred) db = firestore.client() # Nifty 500 Stocks List (Aap yahan apne saare stocks ki list rakh sakte hain) nifty_500_symbols = [ "RELIANCE.NS", "TCS.NS", "HDFCBANK.NS", "INFY.NS", "ICICIBANK.NS", "SBIN.NS", "BHARTIARTL.NS", "LICI.NS", "ITC.NS", "HINDUNILVR.NS", "AEGISVOPAK.NS", "AMBER.NS", "ANGELONE.NS", "ACE.NS", "AIAENG.NS", # ... (Aapki baaki bachi hui Nifty 500 stocks ki list) ] def run_institutional_scanner(): scanned_stocks = [] today_date = datetime.now().strftime('%Y-%m-%d') print(f"🚀 Scanning {len(nifty_500_symbols)} stocks for Institutional Terminal...") for symbol in nifty_500_symbols: try: stock = yf.Ticker(symbol) df = stock.history(period="6mo") if df.empty or len(df) < 50: continue current_price = float(df['Close'].iloc[-1]) prev_price = float(df['Close'].iloc[-2]) # --------------------------------------------------------- # FILTER 1: Price ₹100 se kam nahi honi chahiye # --------------------------------------------------------- if current_price < 100.0: continue current_volume = float(df['Volume'].iloc[-1]) avg_volume_20 = float(df['Volume'].rolling(window=20).mean().iloc[-1]) volume_ratio = round(current_volume / avg_volume_20, 2) if avg_volume_20 > 0 else 1.0 change_val = round(((current_price - prev_price) / prev_price) * 100, 2) # Moving Averages & Range Calculations df['DMA50'] = df['Close'].rolling(window=50).mean() dma_50 = float(df['DMA50'].iloc[-1]) if not pd.isna(df['DMA50'].iloc[-1]) else current_price high_30d = float(df['Close'].tail(30).max()) low_30d = float(df['Close'].tail(30).min()) range_val = round(((high_30d - low_30d) / current_price) * 100, 2) near_high_val = round(((high_30d - current_price) / current_price) * 100, 2) # Dynamic Smart Score Generation (0 to 100) score = 50 if volume_ratio > 1.5: score += 15 if volume_ratio > 3.0: score += 10 if change_val > 0: score += 10 if current_price > dma_50: score += 15 smart_score = min(score, 98) # Category & Reason Assignment if volume_ratio >= 2.5: category = "Intraday" reason = f"High Volume Spike ({volume_ratio}x vs Avg)" elif near_high_val <= 2.0: category = "Swing" reason = "Testing 30D High Breakout Zone" else: category = "Short-Term" reason = "Stable Institutional Accumulation" stock_info = stock.info stock_data = { "symbol": symbol.replace(".NS", ""), "name": stock_info.get('longName', symbol.replace(".NS", "")), "price": f"Rs. {current_price:,.2f}", "change": f"{'+' if change_val >= 0 else ''}{change_val}%", "change_val": float(change_val), "volume_ratio": float(volume_ratio), "volume_spike": f"{volume_ratio}x Avg", "smart_score": int(smart_score), "category": category, "why_selected": reason, "consolidation_range": f"{range_val}% Range", "range_val": float(range_val), "near_high_val": float(near_high_val), "scan_date": today_date } scanned_stocks.append(stock_data) except Exception as e: print(f"Error processing {symbol}: {e}") # Firebase par saare filtered stocks save karein if scanned_stocks: db.collection('daily_picks').document('latest_scan').set({ "date": today_date, "stocks": scanned_stocks, "total_count": len(scanned_stocks) }) print(f"🎉 Success! Uploaded {len(scanned_stocks)} filtered institutional stocks to Firebase.") else: print("⚠️ No stocks matched criteria.") if __name__ == "__main__": run_institutional_scanner()
